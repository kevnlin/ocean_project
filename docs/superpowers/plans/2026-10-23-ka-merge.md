```python
def score_covariance(queries: Tensor, keys: Tensor) -> Tensor:
    """DKNN's covariance of points ``[..., A, D]``: scaled dot products ``q_a . k_b / sqrt(D)``,
    with every point's own score removed except the first (the query's prior variance)."""
    scores = queries @ keys.transpose(-1, -2) / math.sqrt(queries.shape[-1])
    own = torch.eye(scores.shape[-1], dtype=torch.bool, device=scores.device)
    own[0, 0] = False
    return scores.masked_fill(own, 0.)
```

```python
import subprocess, sys, torch, importlib.util
def load(src, name):
    spec = importlib.util.spec_from_loader(name, loader=None)
    m = importlib.util.module_from_spec(spec); exec(compile(src, name, "exec"), m.__dict__); return m
show = lambda ref: subprocess.check_output(["git", "show", f"{ref}:src/ocean_tokenizer/kriging_attention.py"], text=True)
merged = load(open("src/ocean_tokenizer/kriging_attention.py").read(), "merged")
for ref, modes in (("ka-dev5", ("ka_oi", "ka_stationary", "ka_flow", "ka_scores")),
                   ("ka-dev8", ("ka_oi", "ka_stationary", "ka_flow", "ka_cluster", "ka_deep", "ka_product", "ka_multiscale"))):
    other = load(show(ref), ref)
    for mode in modes:
        ell, gamma = torch.full((2, 20), 300.), torch.full((2, 20), .1)
        torch.manual_seed(0); a = merged.KrigingAttention(64, mode, ell, gamma)
        torch.manual_seed(0); b = other.KrigingAttention(64, mode, ell, gamma)
        assert a.state_dict().keys() == b.state_dict().keys(), (ref, mode)
        assert all(torch.equal(a.state_dict()[n], b.state_dict()[n]) for n in a.state_dict()), (ref, mode)
        g = torch.Generator().manual_seed(1)
        q, k = 5, 32
        args = (torch.randn(q, 64, generator=g), torch.randn(q, k, 64, generator=g), 300 * torch.randn(q, k, 2, generator=g),
                torch.randn(q, k, 2, generator=g), torch.rand(q, k, 2, generator=g) > .2, torch.randint(0, 20, (q,), generator=g))
        for p in list(a.parameters()):
            torch.nn.init.normal_(p, 0, .05) if p.ndim > 1 else None
        b.load_state_dict(a.state_dict())
        ya, yb = a(*args), b(*args)
        assert torch.allclose(ya["estimate"], yb["estimate"]) and torch.allclose(ya["variance"], yb["variance"]), (ref, mode)
        print(ref, mode, "ok")
```
